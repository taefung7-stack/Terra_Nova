    // Plays the per-char fade-up stagger every time the section enters

      var parent = document.querySelector(parentSelector);
      if (!parent) return;

      var pio = ('IntersectionObserver' in window) ? new IntersectionObserver(function (ents) {
        ents.forEach(function (e) {

              setTimeout(function () { c.classList.add('in'); }, i * delayStep);

            pio.unobserve(parent);

        });

    }

        var out = '';

          out += '<span class="char" style="transition-delay:' + (charIndex * 70) + 'ms">' + ch + '</span>';
          charIndex++;

    //
    // Why the transition-property dance below: each .char carries an inline
    // `transition-delay: ${i*70}ms` (set during the wrap step). On re-entry,

    // (visible), so the browser elides the transition. Result: the first
    // few chars animate every cycle, the late ones never re-animate.
    //
    // Fix: temporarily set inline `transition-property: none` on each char,
    // remove .in (chars snap invisible without playing the delayed reverse),
    // flush layout, restore transition-property, flush again, then add .in.
    // Now every char starts from a hard-invisible base and the inline delay
    // drives a clean forward stagger — every entry, every char, every time.

          chars.forEach(function (c) { c.style.transitionProperty = 'none'; });
          // 2. Snap to invisible — no animation because property is "none"
          el.classList.remove('in');
          void el.offsetWidth;
          // 3. Restore transitions
          chars.forEach(function (c) { c.style.transitionProperty = ''; });
          void el.offsetWidth;
          // 4. Trigger forward stagger
          el.classList.add('in');

        statementWorries.forEach(function (c) { c.classList.add('in'); });

  // ---------- Hero fallback activator ----------
  // Desktop hero headline is rendered by Three.js. If WebGL fails to init
  // (Chrome's WebGL context limit, hardware accel off, script load error,

  // tagline + "ENGLISH" + "CLICK TERRA NOVA" hint with no actual headline —
  // visually broken. This activator force-displays the static h1 fallback
  // and hides the WebGL-dependent overlay text so the page is always
  // legible regardless of WebGL state.
  function activateHeroFallback() {
    var hero = document.getElementById('hero');
    if (!hero) return;
    hero.classList.add('webgl-failed');
    var fb = document.querySelector('.hero-fallback');
    if (fb) fb.style.display = 'flex';
    ['.hero-tagline', '.hero-english', '#terra-hint'].forEach(function (sel) {
      var el = document.querySelector(sel);
      if (el) el.style.display = 'none';
    });
  }

  // ---------- Three.js scene (Lusion-style cylinder crosses + TERRA NOVA + Big Bang) ----------
  function initScene() {
    if (typeof THREE === 'undefined') { activateHeroFallback(); return; }
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    if (window.innerWidth < 720) return;

    var canvas = document.getElementById('bg-canvas');
    if (!canvas) { activateHeroFallback(); return; }

    var hero = document.getElementById('hero');
    var hintEl = document.getElementById('terra-hint');
    var flashEl = document.getElementById('bigbang-flash');
    var haloEl = document.getElementById('mouse-halo');
    // DOM elements that parallax with the WebGL TERRA NOVA group
    var taglineEl = document.querySelector('.hero-tagline');
    var englishEl = document.querySelector('.hero-english');

    var w = hero.clientWidth;
    var h = hero.clientHeight;

    var renderer;
    try {
      renderer = new THREE.WebGLRenderer({
        canvas: canvas,
        antialias: false,
        alpha: true,
        powerPreference: 'high-performance'
      });
    } catch (e) {
      // WebGL context creation failed — likely the browser's per-tab
      // context limit was hit (often happens during rapid refreshes).
      activateHeroFallback();
      return;
    }
    if (!renderer || !renderer.getContext()) {
      activateHeroFallback();
      return;
    }
    // If the WebGL context is later lost (GPU reset, tab backgrounded too
    // long), gracefully degrade to the static fallback rather than freeze.
    canvas.addEventListener('webglcontextlost', function (ev) {
      ev.preventDefault();
      activateHeroFallback();
    }, { once: true });

    // DPR cap reduced 1.25 → 1.0: with 51 crosses × ~80K-vertex shared mesh,
    // fragment shader cost is the bottleneck on Retina displays during the
    // converge phase. 1.0 cuts pixel work ~36% on hi-DPI screens, restoring
    // 60fps during cross spawn and bigbang gather. Crispness loss is
    // imperceptible against the starfield/bloom backdrop.
    renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, 1.0));
    renderer.setSize(w, h, false);
    renderer.setClearColor(0x000000, 0);

    var scene = new THREE.Scene();
    var camera = new THREE.PerspectiveCamera(50, w / h, 0.1, 100);
    camera.position.set(0, 0, 12);
    camera.lookAt(0, 0, 0);

    // Lights — strong key + mint rim for matte cross surfaces
    scene.add(new THREE.AmbientLight(0xffffff, 0.35));
    var keyLight = new THREE.DirectionalLight(0xffffff, 1.0);
    keyLight.position.set(5, 8, 6);
    scene.add(keyLight);
    var rimLight = new THREE.DirectionalLight(0x2DD4BF, 0.85);
    rimLight.position.set(-4, -2, 4);
    scene.add(rimLight);

    function rand(min, max) { return Math.random() * (max - min) + min; }

    // ---- Materials: factory accepting (color, finish) → mixed glossy/matte per cross ----
    function makeMat(hex, finish) {
      // finish: 'glossy' (high metal, low roughness) or 'matte' (low metal, high roughness)
      if (finish === 'matte') {
        return new THREE.MeshStandardMaterial({ color: hex, metalness: 0.08, roughness: 0.78 });
      }
      return new THREE.MeshStandardMaterial({ color: hex, metalness: 0.55, roughness: 0.22 });
    }

    // ---- Anchor colors (Lusion's "fixed white/black") ----
    // Anchors stay these colors regardless of palette cycle.
    var ANCHOR_WHITE = 0xF0F0F0;
    var ANCHOR_DARK  = 0x111111; // near-pure black, like Lusion

    // ---- Variable palettes — 5 distinct hue families. Click cycles. ----
    // No mint overlap in non-mint palettes → color shift is dramatic & legible.
    var PALETTES = [
      { name: 'mint',   colors: [0x2DD4BF, 0x14B8A6, 0x5EEAD4, 0x99F6E4] }, // P0: mint family (default)
      { name: 'cool',   colors: [0x60A5FA, 0x4FC3F7, 0x7DD3FC, 0x93C5FD] }, // P1: pure blue/cyan
      { name: 'warm',   colors: [0xFCA5A5, 0xFB7185, 0xFFB088, 0xF87171] }, // P2: pure coral/salmon
      { name: 'regal',  colors: [0xC4B5FD, 0xA78BFA, 0xD8B4FE, 0xDDD6FE] }, // P3: pure violet
      { name: 'golden', colors: [0xFDE68A, 0xFCD34D, 0xFFC36F, 0xF59E0B] }  // P4: pure amber/gold
    ];
    var paletteIdx = 0;

    function pickFinish() {
      // 60% glossy, 40% matte — Lusion's mix
      return Math.random() < 0.6 ? 'glossy' : 'matte';
    }
    function pickColorFromPalette(weights /* unused — random pick */) {
      var pal = PALETTES[paletteIdx].colors;
      return pal[Math.floor(Math.random() * pal.length)];
    }

    // ---- Cross prototype: GLTF-loaded merged mesh (Spline export) ----
    // Source: assets/3d/cross.gltf (Spline-exported subdivision mesh, ~80K verts).
    // Prototype is normalized so bounding sphere radius = 1.68 → matches existing
    // spawn box / collision physics (no other constants need to change).
    // window.__crossPrototype is populated by the boot loader before initScene runs.
    var crossPrototype = window.__crossPrototype || null;

    // Procedural fallback (used only if GLTF failed to load) — preserves prior look.
    var capsuleGeom = new THREE.CapsuleGeometry(0.58, 2.20, 6, 14);
    var sphereGeom  = new THREE.SphereGeometry(0.62, 16, 12);

    function makeCross(initialMat) {
      var grp = new THREE.Group();
      if (crossPrototype) {
        // Single merged mesh — geometry is shared across all 51 instances (1 VBO on GPU).
        // Material is per-mesh so palette cycle can independently re-color each cross.
        grp.add(new THREE.Mesh(crossPrototype, initialMat));
      } else {
        // Fallback: 3 capsules + sphere (the previous procedural cross).
        grp.add(new THREE.Mesh(sphereGeom, initialMat));
        grp.add(new THREE.Mesh(capsuleGeom, initialMat));        // Y-axis
        var bx = new THREE.Mesh(capsuleGeom, initialMat);
        bx.rotation.z = Math.PI / 2; grp.add(bx);                 // X-axis
        var bz = new THREE.Mesh(capsuleGeom, initialMat);
        bz.rotation.x = Math.PI / 2; grp.add(bz);                 // Z-axis
      }
      return grp;
    }

    // ---- Spawn: rejection sampling so crosses don't initially overlap ----
    var crosses = [];
    var COUNT = 51;
    var ANCHOR_COUNT = 18; // 8 white + 10 dark — strong fixed presence (Lusion-style)
    var BOX = { x: 10.0, y: 5.8, z: 2.0 };
    var MAX_TRIES = 250;
    var OVERLAP_FACTOR = 0.78; // 22% overlap allowed for visual density

    // Center sub-box: forces some crosses to spawn over the TERRA NOVA Sprite area
    // so the brand text is partially obscured before any mouse interaction.
    var CENTER_BOX = { x: 4.5, y: 1.6, z: 1.5 };

    function trySpawn(scale, boxOverride) {
      var rThis = scale * 1.68; // bounding sphere radius (arm half-length L/2=1.10 + cap r=0.58)
      var bb = boxOverride || BOX;
      for (var tr = 0; tr < MAX_TRIES; tr++) {
        var px = rand(-bb.x, bb.x);
        var py = rand(-bb.y, bb.y);
        var pz = rand(-bb.z, bb.z);
        var ok = true;
        for (var k = 0; k < crosses.length; k++) {
          var c = crosses[k];
          var dx = px - c.position.x, dy = py - c.position.y, dz = pz - c.position.z;
          var minD = (rThis + c.userData.r) * OVERLAP_FACTOR;
          if (dx*dx + dy*dy + dz*dz < minD*minD) { ok = false; break; }
        }
        if (ok) return { x: px, y: py, z: pz, r: rThis };
      }
      // Fallback: random position
      return { x: rand(-bb.x, bb.x), y: rand(-bb.y, bb.y), z: rand(-bb.z, bb.z), r: rThis };
    }

    function setCrossMaterials(cross, hex, finish) {
      var mat = makeMat(hex, finish);
      cross.userData.color = hex;
      cross.userData.finish = finish;
      cross.children.forEach(function (m) {
        if (m.isMesh) {
          if (m.material && m.material.dispose) m.material.dispose();
          m.material = mat;
        }
      });
    }

    function pickRayHit(nx, ny) {
      ndc.set(nx, ny);
      raycaster.setFromCamera(ndc, camera);
      var pool = [terraNova];
      for (var k = 0; k < crosses.length; k++) pool.push(crosses[k]);
      var hits = raycaster.intersectObjects(pool, true);
      if (!hits.length) return null;
      // Walk up to find the cross-Group or terra Sprite
      var first = hits[0];
      if (first.object === terraNova) return { kind: 'terra' };
      var n = first.object;
      while (n) {
        if (n.userData && n.userData.isCross) return { kind: 'cross', target: n };
        n = n.parent;
      }
      return { kind: 'cross' }; // fallback
    }

    function onMove(e) {
      var rect = canvas.getBoundingClientRect();
      targetMouseX = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      targetMouseY = -(((e.clientY - rect.top) / rect.height) * 2 - 1);

      // Mouse halo div follows cursor (idle only)
      if (haloEl && !bigBangTriggered) {
        haloEl.style.left = (e.clientX - rect.left) + 'px';
        haloEl.style.top  = (e.clientY - rect.top) + 'px';
        if (!haloEl.classList.contains('active')) haloEl.classList.add('active');
      }

      if (bigBangTriggered) {
        canvas.classList.remove('is-hover-target');
        return;
      }
      var hit = pickRayHit(targetMouseX, targetMouseY);
      var inCenterZone = (targetMouseX * targetMouseX + targetMouseY * targetMouseY) < 0.18 * 0.18;
      if ((hit && hit.kind === 'terra') || inCenterZone) {
        canvas.classList.add('is-hover-target');
        terraNova.userData.targetOpacity = 1.0;
      } else {
        canvas.classList.remove('is-hover-target');
        terraNova.userData.targetOpacity = 0.85;
      }
    }

    function onClick(e) {
      if (bigBangTriggered) return;

      var nx = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      var ny = -(((e.clientY - rect.top) / rect.height) * 2 - 1);

      // Big-bang trigger zone: ray-hit TERRA NOVA, OR click within central radius
      // (denser cross packing means center pixels are often occluded — accept
      // central clicks as TERRA NOVA intent so users aren't trapped by occlusion)
      var inCenterZone = (nx * nx + ny * ny) < 0.18 * 0.18; // ~18% of half-viewport radius
      if ((hit && hit.kind === 'terra') || inCenterZone) {

        dismissHint();

    // ---- Palette cycle (anchors stay fixed; variable crosses rotate hue) ----
    function cyclePalette() {
      if (bigBangTriggered) return;
      paletteIdx = (paletteIdx + 1) % PALETTES.length;
      crosses.forEach(function (c) {
        if (!c.userData.isAnchor) {
          // Re-color variable crosses, keep finish (avoid harsh visual flicker)
          var newHex = pickColorFromPalette();
          setCrossMaterials(c, newHex, c.userData.finish || pickFinish());
        }
        // Pulse on every cross including anchors (subtle "wave swept" feel)
        var t0 = performance.now();
        var startScale = c.scale.x;
        var peak = startScale * (c.userData.isAnchor ? 1.03 : 1.07);
        (function pulse() {
          var dt = (performance.now() - t0) / 160;
          if (dt < 1) {
            var s = startScale + (peak - startScale) * Math.sin(dt * Math.PI);
            c.scale.set(s, s, s);
            requestAnimationFrame(pulse);
          } else {
            c.scale.set(startScale, startScale, startScale);
          }
        })();
      });
    }

    // ---- Big Bang sequence (converge → hold → flash → dissolve+burst → decel → reveal) ----
    // Timeline:
    //   0     ~  900ms : crosses converge to center, shrinking to point-size (~0.05)
    //    900ms         : warp/rays fade-in begins (.4s transition) — visible WITH the bang
    //    900 ~ 1030ms : HOLD — crosses held at point-size, spinning ("응축, 긴장감")
    //    990ms         : climax flash starts (40ms before the bang for snap)
    //   1030ms         : "펑" — dissolve starts AND star burst starts simultaneously
    //                    (no gap — burst was previously delayed 80ms which read as
    //                     "light orb, then stars" sequence; now they're unified)
    //   1030 ~ 1100ms : DISSOLVE — point-sized crosses fade out
    //   1030 ~ 1090ms : PointLight peak hold (60ms — was 200ms; brief so the orb
    //                    doesn't dwell while the user waits for stars to appear)
    //   1090 ~ 2400ms : PointLight decays, stars stream at peak then settle
    //   2400ms        : reveal hero (TERRA NOVA forward + UI fade in)
    // Convergence + burst — 700ms linear ease. Tuned for "snappy gather":
    // long enough to read as a deliberate inward pull (vs. instant
    // teleport at the previous 480ms), short enough to feel responsive
    // (vs. the draggy 1400ms test). BB PointLight stays dim through
    // ~80% of converge so crosses remain visible during the gather, then
    // spikes hard for the bang.
    //
    // REVEAL_MS pulled forward 2200 → 1600 per user — TERRA NOVA Sprite
    // + site-nav surface 820ms after the bang (was 1420ms; the longer
    // delay made the explosion feel like it stalled before the brand
    // returned). Now the brand reads as the *result* of the bang, not
    // an aftermath several beats later.
    var CONVERGE_MS = 700;
    var HOLD_MS     = 80;
    var CLIMAX_MS   = CONVERGE_MS + HOLD_MS;   // 780ms — moment of "펑"
    var REVEAL_MS   = 1600;

    function triggerBigBang() {
      bigBangTriggered = true;
      // Don't set inline cursor — keep CSS custom cursor (--tn-cursor-default) active
      dismissHint();
      if (haloEl) haloEl.classList.remove('active');
      // NOTE: hero.classList.add('is-bigbang') is deferred to the burst moment
      // (CLIMAX_MS + BURST_DELAY) so the warp canvas + light rays appear AS the
      // stars rush out — not during the converge phase, which previously made
      // the streaks visible while crosses were still gathering (felt premature).

      var bbT0 = performance.now();
      // Snapshot pre-bang state on each cross + clone materials for independent opacity
      crosses.forEach(function (s) {
        s.userData.bbT0 = bbT0;
        s.userData.preBangPos = s.position.clone();
        s.userData.preBangScale = s.scale.x;
        s.children.forEach(function (m) {
          if (m.isMesh && m.material) {
            m.material = m.material.clone();
            m.material.transparent = true;
          }
        });
      });

      // Fade out background TERRA NOVA during convergence
      var fadeStart = performance.now();
      (function fadeOut() {
        var dt = (performance.now() - fadeStart) / 480;
        terraNova.material.opacity = 0.85 * Math.max(0, 1 - dt);
        if (dt < 1) requestAnimationFrame(fadeOut);
        else terraNova.visible = false;
      })();

      // Bigbang PointLight — TWO-STAGE intensity:
      //   • CONVERGE phase: dim ambient glow (0 → 0.6) so crosses stay
      //     visible while gathering. Was building to 5.5 here, drowning
      //     the converge motion in light.
      //   • CLIMAX phase: rapid spike (0.6 → 5.5) over the last ~120ms
      //     of converge + the hold, so the bang lands as a hard flash.
      //   • DECAY: linear fade to 0 over REVEAL_MS.
      bigbangLight = new THREE.PointLight(0xE6F7F4, 0, 26, 1.4);
      bigbangLight.position.set(0, 0, 0);
      scene.add(bigbangLight);
      var BB_PEAK = 5.5;
      var BB_HOLD = 90;
      var BB_SPIKE_START = CONVERGE_MS - 100;   // last 100ms of converge
      (function animateBBLight() {
        var dt = performance.now() - bbT0;
        if (dt < BB_SPIKE_START) {
          // Dim ambient buildup during the visible converge phase.
          var t = dt / BB_SPIKE_START;
          bigbangLight.intensity = 0.6 * (t * t);
        } else if (dt < CLIMAX_MS) {
          // Rapid spike from 0.6 → BB_PEAK across last 120ms+hold.
          var t2 = (dt - BB_SPIKE_START) / (CLIMAX_MS - BB_SPIKE_START);
          bigbangLight.intensity = 0.6 + (BB_PEAK - 0.6) * (t2 * t2);
        } else if (dt < CLIMAX_MS + BB_HOLD) {
          bigbangLight.intensity = BB_PEAK;
        } else if (dt < REVEAL_MS) {
          bigbangLight.intensity = BB_PEAK * (1 - (dt - CLIMAX_MS - BB_HOLD) / (REVEAL_MS - CLIMAX_MS - BB_HOLD));
        } else {
          bigbangLight.intensity = 0;
          return;
        }
        requestAnimationFrame(animateBBLight);
      })();

      // Climax flash (timed slightly before climax for snap)
      setTimeout(function () {
        if (flashEl) {
          flashEl.classList.add('active');
          setTimeout(function () { flashEl.classList.remove('active'); }, 520);
        }
      }, CLIMAX_MS - 40);

      // Warp + rays activation moved from CONVERGE_MS to CLIMAX_MS — they pop on
      // exactly when the burst fires (synced same-frame). Previously is-bigbang
      // fired at CONVERGE_MS=700ms with a .4s transition, leaving warp at ~20%
      // opacity when burst started at 780ms — stars appeared faded/delayed.
      // Now warp + stars + flash all peak together for a true "펑" moment.
      setTimeout(function () {
        hero.classList.add('is-bigbang');
      }, CLIMAX_MS);
      // Star burst at the bang ("펑") — synchronized with cross dissolve so
      // stars rush outward exactly as the point-crosses vanish. No delay.
      setTimeout(function () {
        if (warpField) warpField.startWithBurst();
      }, CLIMAX_MS);
      // Cross husks removed right after dissolve (CLIMAX_MS + 70ms dissolve + 50ms safety = 150ms)
      setTimeout(function () {
        crosses.forEach(function (s) { scene.remove(s); });
        crossesGone = true;
      }, CLIMAX_MS + 150);
      // Reveal hero (TERRA NOVA forward + UI fade in)
      setTimeout(revealHero, REVEAL_MS);
    }

    function spawnOrbs() {
      var orbGeom = new THREE.SphereGeometry(0.65, 16, 16);
      var ORB_COUNT = 60;
      for (var i = 0; i < ORB_COUNT; i++) {
        var isWhite = Math.random() < 0.30;
        // Glassy chrome marble material — low emissive, high metallic, low roughness.
        // Lighting (key + rim + bigbang PointLight) creates true 3D shading.
        var mat = new THREE.MeshStandardMaterial({
          color: isWhite ? 0xffffff : 0x2DD4BF,
          emissive: isWhite ? 0xE6F7F4 : 0x2DD4BF,
          emissiveIntensity: isWhite ? 0.55 : 0.40,
          metalness: 0.70,
          roughness: 0.18,
          transparent: true,
          opacity: 1.0
        });
        var orb = new THREE.Mesh(orbGeom, mat);
        orb.position.set(rand(-0.3, 0.3), rand(-0.3, 0.3), rand(-0.3, 0.3));
        orb.scale.setScalar(0.5); // start small for impact pulse
        orb.userData.spawnTime = performance.now();
        // Spherical-ish random direction with reduced Z spread
        var theta = Math.random() * Math.PI * 2;
        var phi = Math.acos(2 * Math.random() - 1);
        var speed = rand(0.25, 0.55);
        orb.userData.velocity = new THREE.Vector3(
          Math.sin(phi) * Math.cos(theta) * speed,
          Math.sin(phi) * Math.sin(theta) * speed,
          Math.cos(phi) * speed * 0.35
        );
        orb.userData.alive = 1.0;
        orb.userData.lifeRate = rand(0.006, 0.011);

        // Trail: Line with N back-positions, additive blending
        var trailLen = 24;
        var tp = new Float32Array(trailLen * 3);
        for (var t = 0; t < trailLen; t++) {
          tp[t*3] = orb.position.x; tp[t*3+1] = orb.position.y; tp[t*3+2] = orb.position.z;
        }
        var tg = new THREE.BufferGeometry();
        tg.setAttribute('position', new THREE.BufferAttribute(tp, 3));
        var tm = new THREE.LineBasicMaterial({
          color: isWhite ? 0xE6F7F4 : 0x2DD4BF,
          transparent: true,
          opacity: 0.7,
          blending: THREE.AdditiveBlending,
          depthWrite: false
        });
        var trail = new THREE.Line(tg, tm);
        orb.userData.trail = trail;
        orb.userData.trailPositions = tp;
        orb.userData.trailLen = trailLen;
        scene.add(orb);
        scene.add(trail);
        orbs.push(orb);
      }
    }

    function loop() {
      if (!running || !heroVisible) return;
      cancelAnimationFrame(rafId);
      rafId = requestAnimationFrame(tick);

    function tick(now) {
      var t = (now - t0) / 1000;

      // Mouse easing — bumped 0.08 → 0.20 for tighter cursor follow.
      // Previous setting felt laggy because the mWorld point trailed the
      // pointer by ~80ms; with 0.20, the impulse zone tracks the cursor
      // visually 1:1 and the crosses respond as the user moves, not after.
      mouseX += (targetMouseX - mouseX) * 0.20;
      mouseY += (targetMouseY - mouseY) * 0.20;
      var mWorldX = mouseX * 7;
      var mWorldY = mouseY * 4;

      if (!bigBangTriggered) {
        // Idle physics — collision-driven only. No self-spin, no bob.
        // Crosses are still until something hits them: mouse OR another cross.
        // Tuned for snappier "조작감":
        //   • SPRING_K  0.0085 → 0.013 (snappier home recoil)
        //   • MOUSE_R   3.6 → 4.4 (larger interaction halo so cursor feels
        //                          like it's "pushing air" through the field)
        //   • MOUSE_K   0.22 → 0.36 (stronger per-frame impulse — crosses
        //                            actually leap when the cursor passes)
        //   • DAMPING   0.94 → 0.915 (slightly less drag → bouncier rebound,
        //                             still settles before vibration)
        var SPRING_K   = 0.013;
        var DAMPING    = 0.915;
        var ANG_DAMPING = 0.92;
        var MOUSE_R    = 4.4;
        var MOUSE_K    = 0.36;
        var COLLIDE_K  = 0.42;   // collision response strength (was 0.36)
        // PAD must be < OVERLAP_FACTOR (0.78) at spawn — otherwise home positions
        // sit inside the collision zone, and spring↔collision oscillate forever (vibration).
        var COLLIDE_PAD = 0.72;
        var COLLIDE_TORQUE = 0.045; // how much collision tweaks rotation
        var SLEEP_VEL_SQ = 0.0009;  // ~|v| < 0.03 considered low velocity
        var SLEEP_STREAK = 7;       // require N consecutive low-vel frames before sleep
                                    // (prevents oscillation peaks from false-sleeping)

        // Step 1: Spring force + mouse impulse
        for (var i = 0; i < crosses.length; i++) {
          var s = crosses[i];
          var home = s.userData.home;
          var vel  = s.userData.velocity;
          // Spring: only meaningful when displaced from home
          vel.x += (home.x - s.position.x) * SPRING_K;
          vel.y += (home.y - s.position.y) * SPRING_K;
          vel.z += (home.z - s.position.z) * SPRING_K;
          // Mouse impulse (collision-like — adds linear + angular velocity)
          var dx = s.position.x - mWorldX;
          var dy = s.position.y - mWorldY;
          var dist = Math.sqrt(dx * dx + dy * dy);
          if (dist < MOUSE_R && dist > 0.0001) {
            var fmag = (MOUSE_R - dist) * MOUSE_K;
            vel.x += (dx / dist) * fmag;
            vel.y += (dy / dist) * fmag;
            // Tangential angular kick (hand of cards effect) — stronger for bouncy feel
            s.userData.angVel.x += fmag * 0.06 * (Math.random() - 0.5);
            s.userData.angVel.y += fmag * 0.06 * (Math.random() - 0.5);
            s.userData.angVel.z += fmag * 0.10;
          }
        }
        // Step 2: Pair-wise collision push (Newton's 3rd law: equal opposite impulse + small torque)
        for (var ai = 0; ai < crosses.length; ai++) {
          var a = crosses[ai];
          for (var bj = ai + 1; bj < crosses.length; bj++) {
            var b = crosses[bj];
            var ddx = a.position.x - b.position.x;
            var ddy = a.position.y - b.position.y;
            var ddz = a.position.z - b.position.z;
            var d2 = ddx*ddx + ddy*ddy + ddz*ddz;
            var minD = (a.userData.r + b.userData.r) * COLLIDE_PAD;
            if (d2 < minD * minD && d2 > 0.0001) {
              var dd = Math.sqrt(d2);
              var pen = (minD - dd) * COLLIDE_K;
              var nxc = ddx / dd, nyc = ddy / dd, nzc = ddz / dd;
              a.userData.velocity.x += nxc * pen;
              a.userData.velocity.y += nyc * pen;
              a.userData.velocity.z += nzc * pen;
              b.userData.velocity.x -= nxc * pen;
              b.userData.velocity.y -= nyc * pen;
              b.userData.velocity.z -= nzc * pen;
              // Cross torque from collision — perpendicular to contact normal
              a.userData.angVel.x += nyc * pen * COLLIDE_TORQUE;
              a.userData.angVel.y += nzc * pen * COLLIDE_TORQUE;
              a.userData.angVel.z += nxc * pen * COLLIDE_TORQUE;
              b.userData.angVel.x -= nyc * pen * COLLIDE_TORQUE;
              b.userData.angVel.y -= nzc * pen * COLLIDE_TORQUE;
              b.userData.angVel.z -= nxc * pen * COLLIDE_TORQUE;
            }
          }
        }
        // Step 3: Damping + integrate + sleep latch
        for (var ii = 0; ii < crosses.length; ii++) {
          var ss = crosses[ii];
          ss.userData.velocity.multiplyScalar(DAMPING);
          ss.position.add(ss.userData.velocity);
          ss.userData.angVel.multiplyScalar(ANG_DAMPING);
          ss.rotation.x += ss.userData.angVel.x;
          ss.rotation.y += ss.userData.angVel.y;
          ss.rotation.z += ss.userData.angVel.z;
          // Sleep: only after N consecutive low-vel frames (prevents peak-displacement
          // false sleep where vel briefly hits 0 mid-oscillation). When latched,
          // adopt current pos as new home so cross settles slightly displaced.
          if (ss.userData.velocity.lengthSq() < SLEEP_VEL_SQ) {
            ss.userData.lowVelStreak = (ss.userData.lowVelStreak || 0) + 1;
            if (ss.userData.lowVelStreak >= SLEEP_STREAK) {
              ss.userData.home.copy(ss.position);
              ss.userData.velocity.set(0, 0, 0);
              ss.userData.angVel.set(0, 0, 0);
              ss.userData.lowVelStreak = 0;
            }
          } else {
            ss.userData.lowVelStreak = 0;
          }
        }
        // TERRA NOVA hover opacity easing
        var cur = terraNova.material.opacity;
        var tgt = terraNova.userData.targetOpacity;
        terraNova.material.opacity += (tgt - cur) * 0.12;
      } else {
        // Bigbang sequence — separated into 3 distinct beats so the user can read
        // each phase: ① converge to point ② hold (응축, 긴장감) ③ flash + dissolve.
        // Star burst (separate code) fires AFTER this dissolve via BURST_DELAY.
        var POINT_SC   = 0.05;   // converged scale — point-like (was 0.25)
        var DISSOLVE_MS = 70;    // tight dissolve so "펑" reads as instant (was 200)
        for (var ei = 0; ei < crosses.length; ei++) {
          var es = crosses[ei];
          if (!es.userData.bbT0) continue;
          var dtBB = now - es.userData.bbT0;
          if (dtBB < CONVERGE_MS) {
            // ① Converge: lerp pre-bang pos → (0,0,0), shrink toward POINT_SC.
            // Linear ease — uniform inward drift visible from frame 1.
            // (Any acceleration curve compressed too much motion into the
            // last ~150ms, which the user couldn't perceive as a "gather".)
            var t = dtBB / CONVERGE_MS;
            var ease = t;   // pure linear
            var sp = es.userData.preBangPos;
            es.position.x = sp.x * (1 - ease);
            es.position.y = sp.y * (1 - ease);
            es.position.z = sp.z * (1 - ease);
            var ss0 = es.userData.preBangScale;
            var newSc = ss0 + (POINT_SC - ss0) * ease;
            es.scale.set(newSc, newSc, newSc);
            // Rotation accelerates as they speed inward (energy gathering)
            var spinTotal = ease * 0.06;
            es.rotation.x += spinTotal;
            es.rotation.y += spinTotal;
            es.rotation.z += spinTotal * 0.6;
          } else if (dtBB < CONVERGE_MS + HOLD_MS) {
            // ② Hold at point-size with subtle micro-pulse — visualizes condensed
            // energy waiting to release. Brief enough that the user feels tension
            // but doesn't get bored.
            var holdT = (dtBB - CONVERGE_MS) / HOLD_MS;
            var pulse = 1 + Math.sin(holdT * Math.PI * 3) * 0.18; // 1.5 oscillations
            var heldSc = POINT_SC * pulse;
            es.scale.set(heldSc, heldSc, heldSc);
            es.position.set(0, 0, 0);
            es.rotation.x += 0.06;
            es.rotation.y += 0.06;
            es.rotation.z += 0.04;
          } else if (dtBB < CONVERGE_MS + HOLD_MS + DISSOLVE_MS) {
            // ③ Dissolve — the "펑" moment. Point-sized cross shrinks + fades to 0
            // very quickly, perfectly aligned with the climax flash.
            var t2 = (dtBB - CONVERGE_MS - HOLD_MS) / DISSOLVE_MS;
            var s2 = POINT_SC * (1 - t2);
            es.scale.set(s2, s2, s2);
            es.rotation.x += 0.12;
            es.rotation.y += 0.12;
            es.children.forEach(function (m) {
              if (m.isMesh && m.material) m.material.opacity = 1 - t2;
            });
          } else {
            es.visible = false;
          }
        }
        // Orbs + trails (scale pulse on spawn → max → drift)
        for (var oi = 0; oi < orbs.length; oi++) {
          var orb = orbs[oi];
          // Scale pulse: 0.5 → 1.0 over 100ms (impact bloom)
          var elapsedSpawn = now - orb.userData.spawnTime;
          if (elapsedSpawn < 100) {
            var pt = elapsedSpawn / 100;
            orb.scale.setScalar(0.5 + 0.5 * pt);
          } else if (orb.scale.x < 1) {
            orb.scale.setScalar(1.0);
          }

          var tp = orb.userData.trailPositions;
          var tl = orb.userData.trailLen;
          // shift trail back, write head
          for (var ti = tl - 1; ti > 0; ti--) {
            tp[ti*3]     = tp[(ti-1)*3];
            tp[ti*3 + 1] = tp[(ti-1)*3 + 1];
            tp[ti*3 + 2] = tp[(ti-1)*3 + 2];
          }
          tp[0] = orb.position.x; tp[1] = orb.position.y; tp[2] = orb.position.z;
          orb.userData.trail.geometry.attributes.position.needsUpdate = true;

          orb.position.add(orb.userData.velocity);
          orb.userData.velocity.multiplyScalar(1.012); // slight outward acceleration for warp feel
          orb.userData.alive -= orb.userData.lifeRate;
          if (orb.userData.alive < 0) orb.userData.alive = 0;
          orb.material.opacity = orb.userData.alive;
          orb.userData.trail.material.opacity = orb.userData.alive * 0.7;
        }
      }

      // Camera parallax — pre-bigbang only (gives crosses some motion).
      // POST-bigbang: camera locked at (0,0,12) so TERRA NOVA Sprite's offset is the
      // sole source of motion → DOM tagline/english can match it exactly (same mouseX).
      var paraX = bigBangTriggered ? 0 : 0.6;
      var paraY = bigBangTriggered ? 0 : 0.4;
      camera.position.x += (mouseX * paraX - camera.position.x) * 0.04;
      camera.position.y += (mouseY * paraY - camera.position.y) * 0.04;
      camera.lookAt(0, 0, 0);
      // Group parallax: TERRA NOVA Sprite + tagline pill + ENGLISH all driven by the
      // SAME smoothed mouseX/Y → guaranteed in lockstep. No secondary easing on TN.
      // DYNAMIC calibration (was hardcoded for viewport_h=900):
      //   PerspectiveCamera FOV=50° at z=12, terraNova at z=0
      //   → visible world height = 2 × 12 × tan(25°) ≈ 11.19 units
      //   → 1 world unit = currentHeroHeight / 11.19 px
      //   So DOM displacement (px) = WebGL displacement (world units) × pxPerUnit
      //   This keeps lockstep exact at ANY viewport size (1080p, 720p, etc.)
      if (bigBangTriggered && terraNova.visible) {
        var tnDx = -mouseX * 0.5;   // world units
        var tnDy = -mouseY * 0.32;  // world units
        terraNova.position.x = tnDx;
        terraNova.position.y = tnDy;
        var pxPerUnit = hero.clientHeight / 11.19;
        // X axis: Three.js world+X = screen right = CSS translateX(+) → SAME sign
        // Y axis: Three.js world+Y = screen UP, but CSS translateY(+) = screen DOWN
        //         → INVERT sign for DOM Y to keep visual lockstep with terraNova
        var domDx = (tnDx * pxPerUnit).toFixed(1) + 'px';
        var domDy = (-tnDy * pxPerUnit).toFixed(1) + 'px';
        if (taglineEl) {
          taglineEl.style.setProperty('--para-x', domDx);
          taglineEl.style.setProperty('--para-y', domDy);
        }
        if (englishEl) {
          englishEl.style.setProperty('--para-x', domDx);
          englishEl.style.setProperty('--para-y', domDy);
        }
      }

      renderer.render(scene, camera);
      if (running && heroVisible) rafId = requestAnimationFrame(tick);
    }

    loop();
  }

  // ---------- 2D warp starfield (ported from index.html) ----------
  // Background canvas that fades in during/after big-bang to create hyperspace warp.
  // White-dominant (~90%) with mint accent (~10%) per user feedback.
  var warpField = null;
  function initWarpField() {
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return null;
    if (window.innerWidth < 720) return null;
    var canvas = document.getElementById('hero-warp');
    if (!canvas) return null;
    var ctx = canvas.getContext('2d');
    var hero = document.getElementById('hero');
    var stars = [];
    var running = false;
    var rafId = 0;
    var speedMul = 1.0;     // ambient speed
    var burstActive = false;
    var burstT0 = 0;
    var BURST_MS = 2200;
    // Peak warp speed bumped 4.2 → 6.0 for a more violent "빵 터지는"
    // outward rush. Stars at near-z reach the edges in a single frame
    // → reads as a true explosion rather than a fast pan.
    var BURST_PEAK = 6.0;

    function resize() {
      var dpr = Math.min(window.devicePixelRatio || 1, 1.25);
      var w = hero.clientWidth, h = hero.clientHeight;
      canvas.width = w * dpr;
      canvas.height = h * dpr;
      canvas.style.width = w + 'px';
      canvas.style.height = h + 'px';
      ctx.setTransform(1, 0, 0, 1, 0, 0);
      ctx.scale(dpr, dpr);
      canvas.lw = w; canvas.lh = h;
    }
    function init() {
      stars = [];
      var cx = canvas.lw / 2, cy = canvas.lh / 2;
      var md = Math.min(canvas.lw, canvas.lh) * 0.3;
      var n = 2600; // bumped 2200 → 2600 for a denser steady-state stream
      for (var i = 0; i < n; i++) {
        var angle = Math.random() * Math.PI * 2;
        var dist = md + Math.random() * Math.max(canvas.lw, canvas.lh);
        // Initial z range narrowed 500-2000 → 50-1500 so the very first frame
        // already shows stars at every depth (including near-camera). Result:
        // the page opens mid-stream — bright streaks visible immediately
        // instead of starting from a sparse far-field that takes ~1s to fill.
        stars.push({
          x: cx + Math.cos(angle) * dist,
          y: cy + Math.sin(angle) * dist,
          z: Math.random() * 1450 + 50,
          r: Math.random() * 2 + 0.8,
          a: Math.random() * 0.5 + 0.5,
          tw: Math.random() * Math.PI * 2,
          sp: Math.random() * 0.003 + 0.001,
          mint: Math.random() < 0.10  // 10% mint, 90% white
        });
      }
    }
    resize(); init();
    var resizeTimer = null;
    window.addEventListener('resize', function () {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(function () { resize(); init(); }, 250);
    });

    // Mouse-driven vanishing point shift (warp tilts toward cursor for parallax feel)
    var mShiftX = 0, mShiftY = 0;
    var mTargetX = 0, mTargetY = 0;
    var SHIFT_AMP = 90; // max px shift of vanishing point
    hero.addEventListener('mousemove', function (e) {
      var rect = hero.getBoundingClientRect();
      var nx = (e.clientX - rect.left) / rect.width - 0.5;  // -0.5..0.5
      var ny = (e.clientY - rect.top) / rect.height - 0.5;
      mTargetX = nx * 2 * SHIFT_AMP;
      mTargetY = ny * 2 * SHIFT_AMP;
    });
    hero.addEventListener('mouseleave', function () {
      mTargetX = 0; mTargetY = 0;
    });

    // Hold left-click → speed-boost the star rain. Release → ease back to normal.
    var boostMul = 1.0;       // smoothed
    var boostTarget = 1.0;    // target (1.0 normal, 2.6 while holding)
    hero.addEventListener('mousedown', function (e) {
      if (e.button === 0) boostTarget = 2.6;
    });
    // Listen on window for mouseup so dragging out of hero still releases
    window.addEventListener('mouseup', function (e) {
      if (e.button === 0) boostTarget = 1.0;
    });
    window.addEventListener('blur', function () { boostTarget = 1.0; });

    function draw() {
      if (!running) return;
      // Ease vanishing-point toward target mouse position
      mShiftX += (mTargetX - mShiftX) * 0.06;
      mShiftY += (mTargetY - mShiftY) * 0.06;
      // Ease boost (mouse held = faster star rain)
      boostMul += (boostTarget - boostMul) * 0.10;
      var w = canvas.lw, h = canvas.lh, cx = w/2 + mShiftX, cy = h/2 + mShiftY;
      var dpr = Math.min(window.devicePixelRatio || 1, 1.25);
      ctx.setTransform(1, 0, 0, 1, 0, 0);
      ctx.scale(dpr, dpr);
      // Burst speed: HOLD at peak for first 35%, then ease-out cubic to 1.0 over remaining 65%.
      // The hold makes the "fast streaming" feel last longer before stars settle.
      if (burstActive) {
        var bt = (performance.now() - burstT0) / BURST_MS;
        if (bt >= 1) { speedMul = 1.0; burstActive = false; }
        else if (bt < 0.35) {
          speedMul = BURST_PEAK;  // hold at peak
        } else {
          var nt = (bt - 0.35) / 0.65; // remap remaining 65% to 0..1
          var eo = 1 - Math.pow(1 - nt, 3); // ease-out cubic
          speedMul = BURST_PEAK + (1.0 - BURST_PEAK) * eo;
        }
      }
      // Effective speed combines burst (or ambient) with click-hold boost
      var effSpeed = speedMul * boostMul;
      // Trail alpha scales with effective speed → streaks lengthen on boost
      var trailAlpha = burstActive ? Math.max(0.06, 0.15 / effSpeed)
                                   : Math.max(0.07, 0.15 / boostMul);
      ctx.fillStyle = 'rgba(4,4,12,' + trailAlpha + ')';
      ctx.fillRect(0, 0, w, h);
      stars.forEach(function (s) {
        s.z -= 1.3 * effSpeed;   // 1.0 → 1.3 (~30% faster ambient flow per user request)
        if (s.z <= 0) {
          // Stagger respawn depths but keep the whole range packed (no big
          // far-field gap) so the steady-state density matches the initial
          // frame and the stream feels continuous without thinning.
          s.z = 400 + Math.random() * 1100;
          var angle = Math.random() * Math.PI * 2;
          var md = Math.min(w, h) * 0.3;
          var dist = md + Math.random() * Math.max(w, h);
          s.x = cx + Math.cos(angle) * dist;
          s.y = cy + Math.sin(angle) * dist;
        }
        var k = 128 / s.z;
        var px = (s.x - cx) * k + cx;
        var py = (s.y - cy) * k + cy;
        var size = s.r * k * 2;
        s.tw += s.sp;
        var alpha = s.a * (0.5 + 0.5 * Math.sin(s.tw)) * Math.min(1, k);
        var rgb = s.mint ? '94,234,212' : '255,255,255';   // mint accent or white
        var glow = s.mint ? '45,212,191' : '79,195,247';   // mint glow or cool blue
        // Streak (warp tail)
        if (size > 0.5) {
          ctx.beginPath();
          ctx.moveTo(px, py);
          ctx.lineTo(px + (cx - s.x) * k * 0.1, py + (cy - s.y) * k * 0.1);
          ctx.strokeStyle = 'rgba(' + rgb + ',' + (alpha * 0.3) + ')';
          ctx.lineWidth = size * 0.5;
          ctx.stroke();
        }
        // Star body
        ctx.beginPath();
        ctx.arc(px, py, size, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(' + rgb + ',' + alpha + ')';
        ctx.fill();
        // Glow for big stars
        if (size > 1.5) {
          ctx.shadowBlur = size * 3;
          ctx.shadowColor = 'rgba(' + glow + ',' + (alpha * 0.8) + ')';
          ctx.beginPath();
          ctx.arc(px, py, size, 0, Math.PI * 2);
          ctx.fillStyle = 'rgba(' + rgb + ',' + alpha + ')';
          ctx.fill();
          ctx.shadowBlur = 0;
        }
      });
      rafId = requestAnimationFrame(draw);
    }

    return {
      start: function () {
        if (running) return;
        running = true;
        rafId = requestAnimationFrame(draw);
      },
      startWithBurst: function () {
        // Hybrid: 85% of stars reset to NEAR-Z + outer XY → "burst from center" feel.
        // Remaining 15% keep their varied z positions → continuous flow during/after burst.
        // (Was 60/40 — bumped to 85/15 for a much denser initial outward rush
        // matching the "빵 터지는" intent.) Result: explosive bang AND uninterrupted stream.
        var cx = canvas.lw / 2, cy = canvas.lh / 2;
        var burstQty = Math.floor(stars.length * 0.85);
        for (var i = 0; i < burstQty; i++) {
          var s = stars[i];
          // Initial z range tightened from 150-1250 → 25-220 so stars are
          // even closer to camera on the first draw → streaks read as
          // instantaneous explosion rather than "stars take a moment".
          s.z = 25 + Math.random() * 195;
          var angle = Math.random() * Math.PI * 2;
          var md = Math.min(canvas.lw, canvas.lh) * 0.25;
          var dist = md + Math.random() * Math.max(canvas.lw, canvas.lh) * 0.5;
          s.x = cx + Math.cos(angle) * dist;
          s.y = cy + Math.sin(angle) * dist;
        }
        speedMul = BURST_PEAK;
        burstActive = true;
        burstT0 = performance.now();
        if (!running) {
          running = true;
          // Force immediate first frame instead of waiting for the next rAF
          // (~16ms gap). Combined with near-z star reset, streaks are visible
          // synchronously with the call → bang and burst feel unified.
          draw();
          rafId = requestAnimationFrame(draw);
        }
      },
      stop: function () {
        running = false;
        cancelAnimationFrame(rafId);
      }
    };
  }

  // ---------- Boot ----------
  function ready(fn) {
    if (document.readyState !== 'loading') fn();
    else document.addEventListener('DOMContentLoaded', fn);
  }

  // ---- GLTF cross prototype loader ----
  // Loads the Spline-exported cross mesh once, normalizes it to bounding radius 1.68
  // (matches existing spawn/physics constants), and stores on window.__crossPrototype.
  // Called BEFORE initScene so makeCross() finds the prototype synchronously.
  function loadCrossPrototype(cb) {
    if (typeof window.GLTFLoader === 'undefined' || typeof window.THREE === 'undefined') {
      cb(null); return;
    }
    if (typeof window.__crossGltfText === 'undefined') {
      console.error('[cross gltf] inline data missing — assets/3d/cross-gltf-data.js not loaded?');
      cb(null); return;
    }
    var loader = new window.GLTFLoader();
    // Use parse() (not load()) — reads inline JS string instead of fetching.
    // Works with both file:// and http:// (no fetch involved).
    loader.parse(window.__crossGltfText, '', function (gltf) {
      try {
        var crossMesh = null;
        gltf.scene.updateMatrixWorld(true);
        // Spline names the merged cross mesh "Cube 2" (misleading — it's the SDF union output).
        gltf.scene.traverse(function (o) {
          if (o.isMesh && o.name === 'Cube 2') crossMesh = o;
        });
        // Fallback: pick largest mesh by vertex count.
        if (!crossMesh) {
          var maxV = 0;
          gltf.scene.traverse(function (o) {
            if (o.isMesh && o.geometry) {
              var v = o.geometry.attributes.position.count;
              if (v > maxV) { maxV = v; crossMesh = o; }
            }
          });
        }
        if (!crossMesh) { cb(null); return; }

        // Bake parent transforms (root scale 0.01 + Group offset + Cube 2 rotation) into geometry.
        var geom = crossMesh.geometry.clone();
        geom.applyMatrix4(crossMesh.matrixWorld);

        // Center on origin so spawn positions / physics treat the cross as origin-centered.
        geom.computeBoundingBox();
        var center = new window.THREE.Vector3();
        geom.boundingBox.getCenter(center);
        geom.translate(-center.x, -center.y, -center.z);

        // Normalize to bounding sphere radius 1.68 (matches old capsule cross's bounding radius
        // → all spawn box / overlap / Big Bang constants stay valid without re-tuning).
        geom.computeBoundingSphere();
        var TARGET_R = 1.68;
        var scale = TARGET_R / geom.boundingSphere.radius;
        geom.scale(scale, scale, scale);
        geom.computeBoundingSphere();
        geom.computeBoundingBox();

        cb(geom);
      } catch (e) {
        console.error('[cross gltf] extract failed', e);
        cb(null);
      }
    }, function (err) {
      console.error('[cross gltf] parse failed', err);
      cb(null);
    });
  }

  ready(function () {
    initReveals();

    // Every page load — including refreshes — runs the full cinematic intro
    // (crosses + click-to-bigbang ceremony). The previous "skip on refresh"
    // optimization was reverted at user request: refreshes should always
    // return to the first-screen experience.

    warpField = initWarpField();
    // Wire warp canvas pause/resume to hero visibility — it's a fixed canvas
    // inside #hero that keeps drawing 60 fps after the user scrolls past,
    // burning measurable CPU on a hidden surface. The Three.js cross loop
    // (above) already gates itself on heroVisible; the warp field had no
    // such gate — the single biggest "scroll feels heavy after the fold"
    // contributor on this page.
    if (warpField && 'IntersectionObserver' in window) {
      var _heroForWarp = document.getElementById('hero');
      var _warpHeroVisible = true;
      if (_heroForWarp) {
        document.addEventListener('visibilitychange', function () {
          if (document.hidden) warpField.stop();
          else if (_warpHeroVisible) warpField.start();
        });
        var wfio = new IntersectionObserver(function (ents) {
          ents.forEach(function (e) {
            _warpHeroVisible = e.isIntersecting;
            if (e.isIntersecting && !document.hidden) warpField.start();
            else warpField.stop();
          });
        }, { threshold: 0 });
        wfio.observe(_heroForWarp);
      }
    }
    if (!warpField) {
      var staticHero = document.getElementById('hero');
      if (staticHero) {
        staticHero.classList.remove('is-pre');
        staticHero.classList.add('is-post');
      }
      document.body.classList.add('bigbang-done');
      // Warp canvas WebGL failed → Three.js boot is also skipped below,
      // so headline would be missing. Show fallback h1.
      activateHeroFallback();
      if (window.location.hash) {
        setTimeout(function () {
          var target = document.querySelector(window.location.hash);
          if (target) target.scrollIntoView();
        }, 0);
        setTimeout(function () {
          var target = document.querySelector(window.location.hash);
          if (target) target.scrollIntoView();
        }, 400);
      }
      return;
    }
    // Defer Three.js boot until THREE + GLTFLoader (ESM module) are ready.
    // Inline GLTF data (cross-gltf-data.js) is checked separately — if missing,
    // makeCross falls back to procedural capsule cross (page still works).
    function tryBoot() {
      if (typeof window.THREE === 'undefined' || typeof window.GLTFLoader === 'undefined') return false;
      if (typeof window.__crossGltfText !== 'undefined') {
        loadCrossPrototype(function (proto) {
          window.__crossPrototype = proto; // null on failure → procedural fallback
          initScene();
        });
      } else {
        console.warn('[cross] inline GLTF data missing — using procedural fallback');
        window.__crossPrototype = null;
        initScene();
      }
      return true;
    }
    if (!tryBoot()) {
      var maxTries = 60;
      var iv = setInterval(function () {
        if (tryBoot()) clearInterval(iv);
        else if (--maxTries <= 0) {
          // THREE / GLTFLoader never loaded (network failure, blocked CDN, etc.)
          // Stop retrying and degrade to the static fallback so the hero isn't
          // left headline-less.
          clearInterval(iv);
          activateHeroFallback();
        }
      }, 100);
    }
  });
})();
